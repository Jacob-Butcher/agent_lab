```python
def load_corpus_dirs(directories: list[Path]) -> list[CorpusDoc]:
    """加载多个目录，后出现的目录覆盖前面目录中的同名 doc_id。

    单个目录内 doc_id 重复仍然报错；不存在的目录会被跳过。
    """
    merged: dict[str, CorpusDoc] = {}
    for directory in directories:
        if not directory.is_dir():
            continue
        for doc in load_corpus(directory):
            merged[doc.doc_id] = doc
    return list(merged.values())
```

```python
from dataclasses import dataclass
from pathlib import Path

from langchain_core.embeddings import Embeddings

from agentlab.models import CorpusDoc
from agentlab.rag.corpus import load_corpus, load_corpus_dirs
from agentlab.rag.store import ChunkStore


@dataclass
class IngestStats:
    added: int = 0
    changed: int = 0
    skipped: int = 0
    deleted: int = 0
    chunks: int = 0


async def ingest_docs(
    docs: list[CorpusDoc],
    store: ChunkStore,
    embeddings: Embeddings,
    batch_size: int = 10,
    reset: bool = False,
) -> IngestStats:
    existing = {} if reset else await store.get_doc_hashes()
    stats = IngestStats()

    for doc in docs:
        if not reset and existing.get(doc.doc_id) == doc.doc_hash:
            stats.skipped += 1
            continue
        if not doc.chunks:
            continue
        if doc.doc_id in existing:
            stats.changed += 1
        else:
            stats.added += 1

        texts = [chunk.content for chunk in doc.chunks]
        vectors: list[list[float]] = []
        for start in range(0, len(texts), batch_size):
            vectors.extend(
                await embeddings.aembed_documents(texts[start : start + batch_size])
            )

        await store.replace_doc(doc.doc_id, doc.doc_hash, list(doc.chunks), vectors)
        stats.chunks += len(doc.chunks)

    current_ids = {doc.doc_id for doc in docs}
    for doc_id in set(existing) - current_ids:
        await store.delete_doc(doc_id)
        stats.deleted += 1

    return stats


async def ingest_corpus(
    corpus_dir: Path,
    store: ChunkStore,
    embeddings: Embeddings,
    batch_size: int = 10,
    reset: bool = False,
) -> IngestStats:
    return await ingest_docs(
        load_corpus(corpus_dir), store, embeddings, batch_size=batch_size, reset=reset
    )


async def ingest_corpus_dirs(
    directories: list[Path],
    store: ChunkStore,
    embeddings: Embeddings,
    batch_size: int = 10,
    reset: bool = False,
) -> IngestStats:
    return await ingest_docs(
        load_corpus_dirs(directories),
        store,
        embeddings,
        batch_size=batch_size,
        reset=reset,
    )
```

```python
import asyncio
import os
import re
import tempfile
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

import frontmatter
from langchain_core.embeddings import Embeddings

from agentlab.config import Settings
from agentlab.models import Chunk, DocInfo
from agentlab.rag.corpus import load_doc, sha256_text
from agentlab.rag.ingest import ingest_docs
from agentlab.rag.splitter import split_markdown
from agentlab.rag.store import ChunkStore

DOC_ID_PATTERN = re.compile(r"^[a-z0-9][a-z0-9._-]{0,63}$")
PREVIEW_CHARS = 200


class KbError(Exception):
    def __init__(self, status: int, detail: str) -> None:
        super().__init__(detail)
        self.status = status
        self.detail = detail


def _iso(mtime: float) -> str:
    return datetime.fromtimestamp(mtime, tz=timezone.utc).isoformat()


class KBManager:
    def __init__(
        self, settings: Settings, store: ChunkStore, embeddings: Embeddings
    ) -> None:
        self.settings = settings
        self.store = store
        self.embeddings = embeddings
        self.corpus_dir = Path(settings.corpus_dir)
        self.upload_dir = Path(settings.upload_dir)
        self.upload_dir.mkdir(parents=True, exist_ok=True)
        self._lock = asyncio.Lock()

    def _scan_dir(self, directory: Path) -> dict[str, Path]:
        result: dict[str, Path] = {}
        if not directory.is_dir():
            return result
        for path in sorted(directory.glob("*.md")):
            post = frontmatter.load(path)
            doc_id = str(post.metadata.get("doc_id") or path.stem)
            if doc_id in result:
                raise KbError(500, f"目录 {directory} 内 doc_id 重复：{doc_id}")
            result[doc_id] = path
        return result

    def _file_index(self) -> dict[str, tuple[Path, str]]:
        index: dict[str, tuple[Path, str]] = {}
        for doc_id, path in self._scan_dir(self.corpus_dir).items():
            index[doc_id] = (path, "corpus")
        for doc_id, path in self._scan_dir(self.upload_dir).items():
            index[doc_id] = (path, "uploads")
        return index

    @staticmethod
    def _read(path: Path) -> tuple[dict[str, Any], str]:
        post = frontmatter.load(path)
        return dict(post.metadata), post.content

    @staticmethod
    def _write_atomic(path: Path, text: str) -> None:
        path.parent.mkdir(parents=True, exist_ok=True)
        fd, tmp = tempfile.mkstemp(
            dir=path.parent, prefix=f".{path.stem}.", suffix=".tmp"
        )
        try:
            with os.fdopen(fd, "w", encoding="utf-8") as handle:
                handle.write(text)
            os.replace(tmp, path)
        finally:
            if os.path.exists(tmp):
                os.unlink(tmp)

    def _upload_path(self, doc_id: str) -> Path:
        return self.upload_dir / f"{doc_id}.md"

    async def _embed(self, chunks: list[Chunk]) -> list[list[float]]:
        texts = [chunk.content for chunk in chunks]
        vectors: list[list[float]] = []
        batch = self.settings.embed_batch_size
        try:
            for start in range(0, len(texts), batch):
                vectors.extend(
                    await self.embeddings.aembed_documents(texts[start : start + batch])
                )
        except Exception as exc:
            raise KbError(502, "向量化失败，请检查 embedding 配置或稍后重试") from exc
        return vectors

    async def _save_index(
        self, doc_id: str, body: str, chunks: list[Chunk], vectors: list[list[float]]
    ) -> None:
        try:
            await self.store.replace_doc(doc_id, sha256_text(body), chunks, vectors)
        except Exception as exc:
            raise KbError(502, "源文件已保存，索引写入失败，重启后会自动重建") from exc

    async def list_docs(self) -> list[dict]:
        infos: list[DocInfo] = await self.store.list_docs()
        files = self._file_index()
        docs: list[dict] = []
        for info in infos:
            entry = files.get(info.doc_id)
            docs.append(
                {
                    "doc_id": info.doc_id,
                    "title": info.title,
                    "chunks": info.chunks,
                    "source": entry[1] if entry else "uploads",
                    "updated_at": _iso(entry[0].stat().st_mtime) if entry else None,
                }
            )
        return docs

    async def get_doc(self, doc_id: str) -> dict:
        entry = self._file_index().get(doc_id)
        if entry is None:
            raise KbError(404, f"文档不存在：{doc_id}")
        path, source = entry
        metadata, body = self._read(path)
        return {
            "doc_id": doc_id,
            "title": str(metadata.get("title") or doc_id),
            "body": body,
            "source": source,
        }

    async def upload(self, filename: str, data: bytes, overwrite: bool = False) -> dict:
        if not filename.lower().endswith(".md"):
            raise KbError(400, "仅支持 .md 文件")
        if len(data) > self.settings.max_upload_bytes:
            raise KbError(413, f"文件超过 {self.settings.max_upload_bytes} 字节限制")
        try:
            text = data.decode("utf-8")
        except UnicodeDecodeError as exc:
            raise KbError(400, "文件必须是 UTF-8 编码") from exc

        post = frontmatter.loads(text)
        doc_id = str(post.metadata.get("doc_id") or Path(filename).stem)
        if not DOC_ID_PATTERN.match(doc_id):
            raise KbError(
                400, "doc_id 只能包含小写字母、数字、点、下划线、短横线，且不超过 64 字符"
            )
        title = str(post.metadata.get("title") or doc_id)
        body = post.content.strip()
        if not body:
            raise KbError(400, "文档正文为空")

        async with self._lock:
            existing = self._file_index().get(doc_id)
            if existing and not overwrite:
                raise KbError(409, f"doc_id 已存在：{doc_id}，是否覆盖？")
            chunks = list(split_markdown(body, doc_id, title))
            if not chunks:
                raise KbError(400, "文档切块后为空")

            post.content = body
            post.metadata["doc_id"] = doc_id
            post.metadata.setdefault("title", title)
            vectors = await self._embed(chunks)
            self._write_atomic(self._upload_path(doc_id), frontmatter.dumps(post))
            await self._save_index(doc_id, body, chunks, vectors)
            return {"doc_id": doc_id, "title": title, "chunks": len(chunks), "source": "uploads"}

    async def save_body(self, doc_id: str, body: str) -> dict:
        body = body.strip()
        if not body:
            raise KbError(400, "正文不能为空")

        async with self._lock:
            entry = self._file_index().get(doc_id)
            if entry is None:
                raise KbError(404, f"文档不存在：{doc_id}")
            metadata, _old = self._read(entry[0])
            title = str(metadata.get("title") or doc_id)
            chunks = list(split_markdown(body, doc_id, title))
            if not chunks:
                raise KbError(400, "正文切块后为空")

            metadata["doc_id"] = doc_id
            metadata.setdefault("title", title)
            post = frontmatter.Post(body)
            post.metadata = metadata
            vectors = await self._embed(chunks)
            self._write_atomic(self._upload_path(doc_id), frontmatter.dumps(post))
            await self._save_index(doc_id, body, chunks, vectors)
            return {"doc_id": doc_id, "chunks": len(chunks), "source": "uploads"}

    async def delete(self, doc_id: str) -> dict:
        async with self._lock:
            uploads = self._scan_dir(self.upload_dir)
            corpus = self._scan_dir(self.corpus_dir)
            upload_path = uploads.get(doc_id)
            if upload_path is not None:
                upload_path.unlink()
                if doc_id in corpus:
                    doc = load_doc(corpus[doc_id])
                    await ingest_docs(
                        [doc],
                        self.store,
                        self.embeddings,
                        batch_size=self.settings.embed_batch_size,
                    )
                    return {"doc_id": doc_id, "result": "reverted_to_builtin"}
                await self.store.delete_doc(doc_id)
                return {"doc_id": doc_id, "result": "deleted_override"}
            if doc_id in corpus:
                raise KbError(403, "内置文档由 git 管理；如需修改请先编辑生成覆盖版本")
            raise KbError(404, f"文档不存在：{doc_id}")
```

```python
import secrets
from typing import Annotated

from fastapi import APIRouter, Depends, File, HTTPException, Request, UploadFile
from pydantic import BaseModel, Field

from agentlab.kb import KBManager, KbError
from agentlab.services import Services


def require_admin(request: Request) -> KBManager:
    services: Services | None = request.app.state.services
    if services is None:
        raise HTTPException(status_code=503, detail="services not ready")
    settings = services.settings
    if not settings.admin_token:
        raise HTTPException(
            status_code=503, detail="知识库管理未启用，请配置 ADMIN_TOKEN"
        )
    token = request.headers.get("X-Admin-Token", "")
    if not secrets.compare_digest(token, settings.admin_token):
        raise HTTPException(status_code=401, detail="无效的管理令牌")
    return services.kb


class SaveBody(BaseModel):
    body: str = Field(min_length=1)


class SearchRequest(BaseModel):
    query: str = Field(min_length=1, max_length=2000)
    top_k: int | None = None
    threshold: float | None = None


def create_kb_router() -> APIRouter:
    router = APIRouter(prefix="/api/kb")
    Admin = Annotated[KBManager, Depends(require_admin)]

    @router.get("/docs")
    async def list_docs(kb: Admin):
        return {"docs": await kb.list_docs()}

    @router.get("/docs/{doc_id}")
    async def get_doc(doc_id: str, kb: Admin):
        return await kb.get_doc(doc_id)

    @router.post("/docs", status_code=201)
    async def upload_doc(
        kb: Admin,
        file: UploadFile = File(...),
        overwrite: bool = False,
    ):
        if not file.filename:
            raise KbError(400, "缺少文件名")
        data = await file.read(kb.settings.max_upload_bytes + 1)
        return await kb.upload(file.filename, data, overwrite=overwrite)

    @router.put("/docs/{doc_id}")
    async def save_doc(doc_id: str, payload: SaveBody, kb: Admin):
        return await kb.save_body(doc_id, payload.body)

    @router.delete("/docs/{doc_id}")
    async def delete_doc(doc_id: str, kb: Admin):
        return await kb.delete(doc_id)

    @router.post("/search")
    async def search(payload: SearchRequest, kb: Admin):
        return await kb.search(
            payload.query, top_k=payload.top_k, threshold=payload.threshold
        )

    return router
```

```html
const TOKEN_KEY = "kb_admin_token";
let editing = null;

function toast(message, isError = false) {
  const el = document.createElement("div");
  el.className = "toast" + (isError ? " error" : "");
  el.textContent = message;
  document.getElementById("toast").appendChild(el);
  setTimeout(() => el.remove(), 4000);
}

function token() { return sessionStorage.getItem(TOKEN_KEY) || ""; }

async function api(path, options = {}) {
  const headers = Object.assign({}, options.headers || {}, { "X-Admin-Token": token() });
  const response = await fetch(path, Object.assign({}, options, { headers }));
  if (response.status === 401) {
    sessionStorage.removeItem(TOKEN_KEY);
    promptToken("令牌无效，请重新输入");
    throw new Error("unauthorized");
  }
  const text = await response.text();
  let data = null;
  try { data = text ? JSON.parse(text) : null; } catch (_) { data = { detail: text }; }
  if (!response.ok) {
    const detail = data && data.detail ? (typeof data.detail === "string" ? data.detail : JSON.stringify(data.detail)) : "HTTP " + response.status;
    throw new Error(detail);
  }
  return data;
}

function promptToken(message) {
  const value = window.prompt(message || "请输入管理令牌 ADMIN_TOKEN", token());
  if (value === null) return false;
  const trimmed = value.trim();
  if (!trimmed) { sessionStorage.removeItem(TOKEN_KEY); return false; }
  sessionStorage.setItem(TOKEN_KEY, trimmed);
  loadDocs();
  return true;
}

function fmtTime(iso) {
  if (!iso) return "—";
  try { return new Date(iso).toLocaleString(); } catch (_) { return iso; }
}

async function loadDocs() {
  try {
    const data = await api("/api/kb/docs");
    const tbody = document.getElementById("docs");
    tbody.textContent = "";
    document.getElementById("doc-count").textContent = "共 " + data.docs.length + " 篇";
    document.getElementById("empty").classList.toggle("hidden", data.docs.length > 0);
    for (const doc of data.docs) {
      const tr = document.createElement("tr");
      for (const value of [doc.doc_id, doc.title, String(doc.chunks)]) {
        const td = document.createElement("td");
        td.textContent = value;
        tr.appendChild(td);
      }
      const sourceTd = document.createElement("td");
      const badge = document.createElement("span");
      badge.className = "badge" + (doc.source === "uploads" ? " upload" : "");
      badge.textContent = doc.source === "uploads" ? "上传" : "内置";
      sourceTd.appendChild(badge);
      tr.appendChild(sourceTd);
      const timeTd = document.createElement("td");
      timeTd.className = "muted";
      timeTd.textContent = fmtTime(doc.updated_at);
      tr.appendChild(timeTd);
      const actions = document.createElement("td");
      actions.className = "row-actions";
      const editBtn = document.createElement("button");
      editBtn.className = "secondary";
      editBtn.textContent = "查看/编辑";
      editBtn.onclick = () => openEditor(doc.doc_id);
      actions.appendChild(editBtn);
      if (doc.source === "uploads") {
        const delBtn = document.createElement("button");
        delBtn.className = "danger";
        delBtn.textContent = "删除";
        delBtn.onclick = () => removeDoc(doc.doc_id);
        actions.appendChild(delBtn);
      }
      tr.appendChild(actions);
      tbody.appendChild(tr);
    }
  } catch (error) {
    if (error.message !== "unauthorized") toast(error.message, true);
  }
}

async function openEditor(docId) {
  try {
    const doc = await api("/api/kb/docs/" + encodeURIComponent(docId));
    editing = docId;
    document.getElementById("editor").classList.remove("hidden");
    document.getElementById("edit-title").textContent = doc.doc_id + " · " + doc.title;
    document.getElementById("edit-body").value = doc.body;
    document.getElementById("edit-source").textContent = doc.source === "uploads" ? "来源：上传" : "来源：内置（保存后生成覆盖版本）";
    document.getElementById("editor").scrollIntoView({ behavior: "smooth" });
  } catch (error) {
    if (error.message !== "unauthorized") toast(error.message, true);
  }
}

async function saveEditor() {
  if (!editing) return;
  const button = document.getElementById("save-btn");
  button.disabled = true;
  try {
    const body = document.getElementById("edit-body").value;
    const result = await api("/api/kb/docs/" + encodeURIComponent(editing), {
      method: "PUT",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ body }),
    });
    toast("已保存 " + result.doc_id + "（" + result.chunks + " 个片段）");
    closeEditor();
    loadDocs();
  } catch (error) {
    if (error.message !== "unauthorized") toast(error.message, true);
  } finally {
    button.disabled = false;
  }
}

function closeEditor() {
  editing = null;
  document.getElementById("editor").classList.add("hidden");
}

async function removeDoc(docId) {
  if (!window.confirm("确定删除 " + docId + "？若存在内置版本会自动回退。")) return;
  try {
    const result = await api("/api/kb/docs/" + encodeURIComponent(docId), { method: "DELETE" });
    toast(result.result === "reverted_to_builtin" ? "已删除覆盖，恢复为内置版本" : "已删除文档");
    loadDocs();
  } catch (error) {
    if (error.message !== "unauthorized") toast(error.message, true);
  }
}

async function uploadFile(file, overwrite) {
  const form = new FormData();
  form.append("file", file);
  const url = "/api/kb/docs" + (overwrite ? "?overwrite=true" : "");
  return api(url, { method: "POST", body: form });
}

async function handleFile(file) {
  if (!file) return;
  if (!file.name.toLowerCase().endsWith(".md")) { toast("仅支持 .md 文件", true); return; }
  try {
    const result = await uploadFile(file, false);
    toast("已上传 " + result.doc_id);
    loadDocs();
  } catch (error) {
    if (error.message === "unauthorized") return;
    if (error.message.indexOf("已存在") !== -1) {
      if (window.confirm(error.message + "，是否覆盖？")) {
        try {
          const result = await uploadFile(file, true);
          toast("已覆盖 " + result.doc_id);
          loadDocs();
        } catch (overwriteError) {
          if (overwriteError.message !== "unauthorized") toast(overwriteError.message, true);
        }
      }
    } else {
      toast(error.message, true);
    }
  }
}

async function runSearch() {
  const query = document.getElementById("q").value.trim();
  if (!query) { toast("请输入问题", true); return; }
  const topK = Number(document.getElementById("topk").value || 5);
  const threshold = Number(document.getElementById("th").value || 0);
  const box = document.getElementById("results");
  box.textContent = "";
  try {
    const data = await api("/api/kb/search", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ query, top_k: topK, threshold }),
    });
    if (!data.results.length) {
      const empty = document.createElement("p");
      empty.className = "muted";
      empty.textContent = "没有命中的片段，试试降低阈值或换个问法";
      box.appendChild(empty);
      return;
    }
    for (const row of data.results) {
      const card = document.createElement("div");
      card.style.border = "1px solid var(--line)";
      card.style.borderRadius = "8px";
      card.style.padding = "10px";
      const head = document.createElement("div");
      const code = document.createElement("code");
      code.textContent = row.doc_id + "#" + row.chunk_index;
      head.appendChild(code);
      const score = document.createElement("span");
      score.className = "score";
      score.textContent = "  " + Number(row.score).toFixed(3);
      head.appendChild(score);
      const meta = document.createElement("span");
      meta.className = "muted";
      meta.textContent = "  " + row.title + (row.heading ? " / " + row.heading : "");
      head.appendChild(meta);
      const content = document.createElement("p");
      content.className = "muted";
      content.textContent = row.content;
      card.appendChild(head);
      card.appendChild(content);
      box.appendChild(card);
    }
  } catch (error) {
    if (error.message !== "unauthorized") toast(error.message, true);
  }
}
```
